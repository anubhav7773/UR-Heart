5. ZERO-LATENCY DOUBLE-BUFFERED AD ENGINE
Waiting for a video ad to load ruins user immersion. UR-Heart implements a FIFO Double-Buffer Preloader:

Background worker hamesha 2 rewarded video ads memory mein ready rakhta hai.

Jab user "Quick Reflection" tap karta hai, primary ad 0-millisecond delay par play hota hai.

Ad play hote hi secondary ad primary ban jata hai, aur background mein next ad load hona shuru ho jata hai.

Dart


// lib/core/ads/rewarded_ad_manager.dart
import 'dart:async';
import 'package:google_mobile_ads/google_mobile_ads.dart';

class RewardedAdManager {
  static final RewardedAdManager instance = RewardedAdManager._internal();
  RewardedAdManager._internal();

  RewardedAd? _primaryBufferAd;
  RewardedAd? _secondaryBufferAd;
  bool _isLoading = false;

  static const String adUnitId = 'ca-app-pub-XXXXXXXXXXXXXXXX/YYYYYYYYYY';

  Future<void> initializePreloader() async {
    await _loadAdIntoBuffer();
    await _loadAdIntoBuffer();
  }

  Future<void> _loadAdIntoBuffer() async {
    if (_isLoading) return;
    if (_primaryBufferAd != null && _secondaryBufferAd != null) return;

    _isLoading = true;
    await RewardedAd.load(
      adUnitId: adUnitId,
      request: const AdRequest(),
      rewardedAdLoadCallback: RewardedAdLoadCallback(
        onAdLoaded: (RewardedAd ad) {
          _isLoading = false;
          if (_primaryBufferAd == null) {
            _primaryBufferAd = ad;
          } else if (_secondaryBufferAd == null) {
            _secondaryBufferAd = ad;
          }
        },
        onAdFailedToLoad: (LoadAdError error) {
          _isLoading = false;
          // Retry with exponential backoff
          Future.delayed(const Duration(seconds: 5), () => _loadAdIntoBuffer());
        },
      ),
    );
  }

  /// Displays buffered ad instantly with cryptographic SSV custom data
  void showRewardedAd({
    required String userId,
    required String adType,
    required String targetId,
    required Function() onRewardGranted,
  }) {
    final adToPlay = _primaryBufferAd ?? _secondaryBufferAd;

    if (adToPlay == null) {
      // Fallback: refill buffer and grant grace allowance if network choked
      _loadAdIntoBuffer();
      return;
    }

    // Attach Server-Side Verification (SSV) Options
    final ssvOptions = ServerSideVerificationOptions(
      customData: '$userId:$adType:$targetId',
    );
    adToPlay.setServerSideVerificationOptions(ssvOptions);

    adToPlay.fullScreenContentCallback = FullScreenContentCallback(
      onAdDismissedFullScreenContent: (ad) {
        ad.dispose();
        if (adToPlay == _primaryBufferAd) {
          _primaryBufferAd = _secondaryBufferAd;
          _secondaryBufferAd = null;
        } else {
          _secondaryBufferAd = null;
        }
        _loadAdIntoBuffer(); // Immediately replenish buffer
      },
      onAdFailedToShowFullScreenContent: (ad, error) {
        ad.dispose();
        _primaryBufferAd = null;
        _loadAdIntoBuffer();
      },
    );

    adToPlay.show(onUserEarnedReward: (ad, reward) {
      onRewardGranted();
    });
  }
}

# app/api/v1/endpoints/billing_webhook.py
import os
from fastapi import APIRouter, Request, HTTPException, status, Depends
from sqlalchemy.ext.asyncio import AsyncSession
from sqlalchemy import update, select
from uuid import UUID
from datetime import datetime
from app.core.database import get_db
from app.models.domain.user import User
from app.models.domain.in_app_purchases import InAppPurchase

router = APIRouter(prefix="/billing", tags=["Billing & Store Webhooks"])

REVENUECAT_SECRET = os.getenv("REVENUECAT_WEBHOOK_SECRET")

@router.post("/revenuecat-webhook", status_code=status.HTTP_200_OK)
async def process_revenuecat_event(request: Request, db: AsyncSession = Depends(get_db)):
    auth_header = request.headers.get("Authorization")
    if auth_header != f"Bearer {REVENUECAT_SECRET}":
        raise HTTPException(status_code=status.HTTP_401_UNAUTHORIZED, detail="Invalid webhook token")

    payload = await request.json()
    event = payload.get("event", {})
    event_type = event.get("type")
    user_id_str = event.get("app_user_id")
    tx_id = event.get("id")

    if not user_id_str or not tx_id:
        return {"status": "ignored"}

    user_uuid = UUID(user_id_str)

    # 1. Idempotency Check: Prevent duplicate credit
    existing = await db.execute(select(InAppPurchase).where(InAppPurchase.transaction_reference == tx_id))
    if existing.scalar_one_or_none():
        return {"status": "already_processed"}

    # 2. Subscription Grant
    if event_type in ("INITIAL_PURCHASE", "RENEWAL"):
        product_id = event.get("product_id", "")
        exp_ms = event.get("expiration_at_ms")
        expires_at = datetime.utcfromtimestamp(exp_ms / 1000.0) if exp_ms else None

        tier = "monthly"
        if "weekly" in product_id:
            tier = "weekly"
        elif "lifetime" in product_id:
            tier = "lifetime"

        # Update User Sovereign Entitlements
        await db.execute(
            update(User)
            .where(User.id == user_uuid)
            .values(
                subscription_tier=tier,
                subscription_expires_at=expires_at,
                is_ad_free=True,
                swipes_remaining=999999,
                direct_letters_count=User.direct_letters_count + 5
            )
        )

        # Audit Record (15% Google Store Cut Calculation)
        gross = float(event.get("price_in_purchased_currency", 14.99))
        fee = round(gross * 0.15, 2)
        net = round(gross - fee, 2)

        iap_record = InAppPurchase(
            user_id=user_uuid,
            transaction_reference=tx_id,
            product_identifier=product_id,
            store="google_play",
            currency=event.get("currency", "USD"),
            amount_gross=gross,
            platform_fee=fee,
            amount_net=net
        )
        db.add(iap_record)
        await db.commit()

    return {"status": "success"}
